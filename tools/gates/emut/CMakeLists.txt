# arch025-emut: libTooling tool linked against the official LLVM release archive (clang+llvm-<v>-x86_64-pc-windows-msvc).
# The archive is built with /MT (CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded), RTTI off and EH off: the tool matches it.
#   cmake -G Ninja -S tools/gates/emut -B <build> -DWHIRL_LLVM=C:/ClaudeWorkspace/tools/llvm/21.1.8   (inside vcvars64)
cmake_minimum_required(VERSION 3.20)
project(emut CXX)
if(NOT WHIRL_LLVM)
  set(WHIRL_LLVM $ENV{WHIRL_LLVM})
endif()
if(NOT WHIRL_LLVM)
  message(FATAL_ERROR "set WHIRL_LLVM to the extracted LLVM release folder")
endif()
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
find_package(Clang REQUIRED CONFIG PATHS "${WHIRL_LLVM}/lib/cmake/clang" NO_DEFAULT_PATH)
# LLVMDebugInfoPDB names the release builder's DIA SDK path (VS 2019 Professional); use this machine's VS DIA SDK
get_target_property(_pdb LLVMDebugInfoPDB INTERFACE_LINK_LIBRARIES)
file(TO_CMAKE_PATH "$ENV{VSINSTALLDIR}" _vs)
file(GLOB _dia "${_vs}/DIA SDK/lib/amd64/diaguids.lib")
list(FILTER _pdb EXCLUDE REGEX "diaguids\.lib$")
if(_dia)
  list(PREPEND _pdb "${_dia}")
endif()
set_target_properties(LLVMDebugInfoPDB PROPERTIES INTERFACE_LINK_LIBRARIES "${_pdb}")
message(STATUS "DIA SDK: ${_dia}")
message(STATUS "LLVM ${LLVM_PACKAGE_VERSION} at ${LLVM_INSTALL_PREFIX}; RTTI ${LLVM_ENABLE_RTTI}; EH ${LLVM_ENABLE_EH}")
add_executable(emut emut.cpp)
# arch025-emut2: embed the sha256 of emut.cpp (CRLF -> LF); a source change re-runs configure, emut_run.py compares it
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/emut.cpp")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/emut.cpp" _emut_src)
string(REPLACE "\r\n" "\n" _emut_src "${_emut_src}")
string(SHA256 _emut_sha "${_emut_src}")
target_compile_definitions(emut PRIVATE EMUT_SRC_SHA256="${_emut_sha}")
message(STATUS "emut.cpp sha256 ${_emut_sha}")
set_target_properties(emut PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_include_directories(emut SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
separate_arguments(_llvm_defs NATIVE_COMMAND "${LLVM_DEFINITIONS}")
target_compile_definitions(emut PRIVATE ${_llvm_defs})
target_compile_options(emut PRIVATE /GR- /EHs-c- /D_HAS_EXCEPTIONS=0 /utf-8 /Zc:__cplusplus /permissive- /W3 /wd4146 /wd4244 /wd4267 /wd4291 /wd4624)
target_link_libraries(emut PRIVATE clangTooling clangASTMatchers clangAnalysis clangFrontend clangDriver clangSerialization
                      clangParse clangSema clangAST clangLex clangBasic)
